local dynamiteobject = game.ServerStorage.dynamite
local bomb = game.ReplicatedStorage.remoteevents.bomb
local functions = require(game.ServerScriptService.progress.functions)
local speed = 130
local gravitymultiplier = 0.1
local maxdistance = 250



bomb.OnServerEvent:Connect(function(player, mousepos)

	local character = player.Character
	if not character then return end

	local hrp = character:FindFirstChild("HumanoidRootPart")
	if not hrp then return end

	local humanoid = character:FindFirstChild("Humanoid")
	local animator = humanoid:FindFirstChild("Animator")


	local throwinganim = game.ReplicatedStorage.animations:WaitForChild("throwing")
	local animtrack = animator:LoadAnimation(throwinganim)
    animtrack.Priority = Enum.AnimationPriority.Action
	
	local originalpos = hrp.Position
	local direction = (mousepos - originalpos).Unit


	if direction:Dot(hrp.CFrame.LookVector) <= 0 then return end 

	
	animtrack:Play()
	task.wait(0.2)

	local dynamite = dynamiteobject:Clone() 
	
	dynamite.CFrame = CFrame.new(originalpos , originalpos + direction) * CFrame.new(0,.7,-2)
	dynamite.Parent = game.Workspace


	dynamite.AssemblyLinearVelocity = direction * speed
    local terrain = game.Workspace:FindFirstChild("terrain"):FindFirstChild("grassterrain")




while dynamite ~= nil do 
task.wait(.3)
	if dynamite.Position.y - terrain.Position.y - terrain.Size.Y / 2  <= 4 then 		


			local explosionvfxremote = game.ReplicatedStorage.remoteevents.explosionremote
			explosionvfxremote:FireAllClients(dynamite.Position)
			dynamite:Destroy()


            local range = 20

            for _ , i in pairs(game.Workspace:GetChildren()) do
                  if i.Name == "rock" then 

                    local distance = (i.Position - dynamite.Position).Magnitude
                    if distance <= range then

                        i:Destroy()
functions.getcurrency(player, "gems" ,5)
						functions.rockspawn()
                    end
end 

end
         

			
			

dynamite = nil

		
	
    
   end
end
end)
